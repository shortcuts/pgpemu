# The bytes come from the firmware response layout and the existing firmware asserts, not from packet captures.
#
# Grammar (whitespace-separated, hex without 0x):
#   opcode <NAME> <hex>                              NAME = CONTROL_OP_<NAME> = Opcode.<NAME>
#   status <NAME> <hex>                              NAME = CONTROL_STATUS_<NAME> = StatusCode.<NAME>
#   uuid <key> <uuid>                                standard UUID string form
#   event <NAME> <hex>                               NAME = PGP_EVENT_<NAME> = EventType.<NAME>
#   vector <name> <setup> <request> <response> [prefix]
#     request  = Command characteristic write: [opcode][payload]
#     response = Response indication: [status][opcode][payload]
#     prefix   = the firmware frame may be longer; only these bytes must match
#
#   vector_event <name> <NAME> <frame>
#     frame = Event characteristic notification [seq][type][slot], emitted for a game link on Device Profile slot 1
#     to a subscribed link whose first event this is
#
